# =============================================================================
# Event 1: Deklaracja Wyrów (1936) - fires on game start for SLS
# =============================================================================
country_event = {
	id = SLS_events.1
	title = SLS_events.1.t
	desc = SLS_events.1.desc
	picture = GFX_report_event_generic_rally

	is_triggered_only = no
	fire_only_once = yes

	trigger = {
		tag = SLS
		date > 1936.1.1
		NOT = { has_country_flag = SLS_deklaracja_done }
	}

	mean_time_to_happen = { days = 5 }

	# Option A: Close trade relations with Poland ("Węzeł Węglowy" buff)
	option = {
		name = SLS_events.1.a
		POL = { add_opinion_modifier = { target = SLS modifier = SLS_alliance_opinion } }
		add_stability = 0.05
		set_country_flag = SLS_deklaracja_done
		set_country_flag = SLS_trade_with_poland
	}
	# Option B: Isolation
	option = {
		name = SLS_events.1.b
		remove_ideas = SLS_wezel_weglowy
		add_political_power = 100
		add_stability = -0.05
		set_country_flag = SLS_deklaracja_done
		set_country_flag = SLS_isolation
	}
}

# =============================================================================
# Event 3: Uchodźcy ze Wschodu i Zachodu (dynamic, war-triggered)
# =============================================================================
country_event = {
	id = SLS_events.3
	title = SLS_events.3.t
	desc = SLS_events.3.desc
	picture = GFX_report_event_generic_refugees

	is_triggered_only = no
	fire_only_once = no

	trigger = {
		tag = SLS
		has_war = no
		OR = {
			POL = { has_war = yes }
			GER = { has_war = yes }
			CZE = { has_war = yes }
		}
		NOT = { has_country_flag = SLS_refugees_recently }
	}

	mean_time_to_happen = { days = 60 }

	# Accept refugees
	option = {
		name = SLS_events.3.a
		add_manpower = 15000
		add_stability = -0.03
		set_country_flag = { flag = SLS_refugees_recently days = 180 value = 1 }
	}
	# Close borders
	option = {
		name = SLS_events.3.b
		add_stability = 0.02
		add_political_power = 25
		set_country_flag = { flag = SLS_refugees_recently days = 180 value = 1 }
	}
}
